import pandas as pd
E="docs/quality/evidence/CR-0021/"
W={"Alder Flycatcher","Common Yellowthroat","Swamp Sparrow","Northern Waterthrush"}
raw=pd.concat([pd.read_csv(f"/home/ec2-user/cr0021_scratch/data/negatives/gbif_negatives_{r}.csv",usecols=["gbif_id","common_name"]) for r in ["ME","NH","VT"]])
sp=dict(zip(raw.gbif_id.astype(str),raw.common_name))
C=pd.read_csv(E+"preregister_C.csv"); N=pd.read_csv(E+"preregister_N.csv")
C["key"]=list(zip(C.longitude.round(6),C.latitude.round(6))); N["key"]=list(zip(N.longitude.round(6),N.latitude.round(6)))
N=N.merge(C[["key","gbif_id"]],on="key",how="left")
N["wet"]=N.gbif_id.astype("Int64").astype(str).map(sp).isin(W)
h=N[~N.is_nonveg]
for r,g in h.groupby("region"):
    base=g[g.year<=2022].wet.mean()*100
    print(r,"2020-22 %.1f%%"%base, " ".join("%d: %.1f%% (%+.1f pp)"%(y,g[g.year==y].wet.mean()*100,g[g.year==y].wet.mean()*100-base) for y in (2023,2024)), "| unmatched", int(g.gbif_id.isna().sum()))
